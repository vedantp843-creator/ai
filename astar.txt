from queue import PriorityQueue

goal = [[1, 2, 3],
        [4, 5, 6],
        [7, 8, 0]]

def heuristic(state):
    count = 0
    for i in range(3):
        for j in range(3):
            if state[i][j] != 0 and state[i][j] != goal[i][j]:
                count += 1
    return count

def find_blank(state):
    for i in range(3):
        for j in range(3):
            if state[i][j] == 0:
                return i, j

def state_to_tuple(state):
    return tuple(tuple(row) for row in state)

def generate_moves(state):
    x, y = find_blank(state)
    directions = [(-1,0),(1,0),(0,-1),(0,1)]
    children = []

    for dx, dy in directions:
        nx, ny = x + dx, y + dy

        if 0 <= nx < 3 and 0 <= ny < 3:
            new_state = [row[:] for row in state]

            new_state[x][y], new_state[nx][ny] = \
                new_state[nx][ny], new_state[x][y]

            children.append(new_state)

    return children

def a_star(start):
    pq = PriorityQueue()

    g = 0
    h = heuristic(start)
    pq.put((g + h, g, start))

    visited = set()

    while not pq.empty():
        f, g, current = pq.get()

        print("\nCurrent State:")
        for row in current:
            print(row)

        if current == goal:
            print("\nGoal Reached!")
            return

        visited.add(state_to_tuple(current))

        for child in generate_moves(current):
            if state_to_tuple(child) not in visited:
                g_new = g + 1
                h_new = heuristic(child)
                pq.put((g_new + h_new, g_new, child))

start = [[1, 2, 3],
         [4, 0, 6],
         [7, 5, 8]]

a_star(start)
